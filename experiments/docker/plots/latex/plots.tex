%% \countplotna{FILE}{COLUMN} sets \plotnacount to the number of rows of the csv file
%% FILE, relative to \resultsdir, in which COLUMN is n/a (\plotna), e.g. the number of
%% benchmarks a tool does not support.
\newcommand{\countplotna}[2]{%
  \pgfplotstableread[col sep=comma]{\resultsdir/#1}\plotnatable
  \pgfplotstablegetrowsof{\plotnatable}%
  \pgfmathtruncatemacro{\plotnalast}{\pgfplotsretval-1}%
  \xdef\plotnacount{0}%
  \foreach \plotnarow in {0,...,\plotnalast}{%
    \pgfplotstablegetelem{\plotnarow}{#2}\of\plotnatable
    \pgfmathtruncatemacro{\plotnanew}{\plotnacount+(\pgfplotsretval==\plotna)}%
    \xdef\plotnacount{\plotnanew}%
  }%
}

%% A quantile plot spanning the full text width.
%% #1: optional axis overrides
%% #2: csv file, relative to \resultsdir
%% #3: comma separated list of column/colour/legend-entry triples
%% #4: the number of benchmarks (=xmax)
%% Further plots without legend entry can be added by locally redefining \quantileplotextras, e.g., within the surrounding minipage.
\newcommand{\quantileplotextras}{}
\def\quantilenan{nan}
\newcommand{\quantileplot}[4][]{%
  \begin{tikzpicture}
    \begin{axis}[
      width=0.9\linewidth,
      height=\quantileplotheight,
      scale only axis,
      xmin=0, xmax=#4,
      ymin=1, ymax=1000,
      ymode=log,
      ymajorgrids,
      axis x line=bottom,
      axis y line=left,
      unbounded coords=discard, filter discard warning=false,
      xlabel={solved benchmarks (out of #4)},
      ylabel={runtime},
      xlabel style={font=\scriptsize, yshift=4pt},
      ylabel style={font=\scriptsize, yshift=-7pt},
      log ticks with fixed point,
      xticklabel style={font=\scriptsize},
      yticklabel style={font=\scriptsize},
      legend pos=south east,
      legend columns=4,
      legend style={font=\fontsize{6.5}{7.5}\selectfont, draw=none, fill=white, fill opacity=0.8, text opacity=1,
                    inner sep=1pt, column sep=2pt, row sep=-1pt,
                    at={(1.02,0.01)}, anchor=south east,
                    /tikz/every even column/.append style={column sep=5pt}},
      /pgfplots/legend image code/.code={%
        \draw[mark repeat=2, mark phase=2, ##1] plot coordinates {(0cm,0cm) (0.25cm,0cm)};},
      every axis plot/.append style={ultra thick},
      legend cell align={left},
      #1,
    ]
      \pgfplotstableread[col sep=comma]{\resultsdir/#2}\quantiletable
      \pgfplotstablegetrowsof\quantiletable
      \edef\quantilerows{\pgfplotsretval}%
      \foreach \col/\colour/\name in {#3}{%
        % A column without any solved benchmark, e.g. of Storm 1.6.2 on ARM, starts
        % with nan; pgfplots would drop its empty plot but keep its legend entry, which
        % then labels the next line, so both are left out
        \ifnum\quantilerows>0
          \pgfplotstablegetelem{0}{\col}\of\quantiletable
          \edef\firstvalue{\pgfplotsretval}%
        \else
          \let\firstvalue\quantilenan
        \fi
        \ifx\firstvalue\quantilenan\else
          \edef\plotline{\noexpand\addplot[\colour]
            table [x=i, y=\col, col sep=comma] {\resultsdir/#2};
            \noexpand\addlegendentry{\unexpanded\expandafter{\name}}}%
          \plotline
        \fi
      }
      \quantileplotextras
    \end{axis}
  \end{tikzpicture}%
}
